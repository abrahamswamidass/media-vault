# ---------------------------------------------------------------------------
# Media Vault — Python dependencies
#
# The CORE harness (nas / archive / amazon connectors + CLI) uses only the
# Python standard library, so this file can be empty and things still run.
# The packages below are for the NEXT features; uncomment as you enable them.
# ---------------------------------------------------------------------------

# --- Metadata extraction (wraps the exiftool binary baked into the image) ---
PyExifTool>=0.5.6

# --- Perceptual dedup (find re-compressed cloud copies of NAS originals) ----
Pillow>=10.0.0
ImageHash>=4.3.1

# --- HEIC/HEIF decoding (the default iPhone photo format since iOS 11) ------
# Plain Pillow can't open these at all — raises "cannot identify image file",
# not a clear "unsupported format" error. imaging.py registers this opener
# automatically the first time it touches Pillow; nothing else to configure.
pillow-heif>=0.15.0

# --- Face detection/clustering (set FACES_LIVE=1 to use) --------------------
# ONNX-based, not dlib — no C++ toolchain needed in this slim Docker image.
# First real run downloads model weights (a few hundred MB, cached after
# that under ~/.insightface) — needs real internet access once.
insightface>=0.7.3
onnxruntime>=1.16.0

# --- Google Drive live connector (set DRIVE_LIVE=1 to use) ------------------
google-api-python-client>=2.100.0
google-auth-oauthlib>=1.2.0
google-auth-httplib2>=0.2.0

# --- NAS direct SMB connector (set NAS_MODE=smb to use) --------------------
# Bypasses the OS mount and talks SMB2/3 directly. Needed when Docker Desktop
# on Windows can't bind-mount a mapped network drive (mounts empty).
smbprotocol>=1.12.0

# --- GCP mirror push (Firestore metadata + Cloud Storage thumbnails) -------
google-cloud-firestore>=2.13.0
google-cloud-storage>=2.12.0
